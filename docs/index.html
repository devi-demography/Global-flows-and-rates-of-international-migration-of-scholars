Data Preparation

Prepare country-level geographical information and convert country codes from ISO-3 to ISO-2.

Code 
countries <- spData::world

scopus_migration <- scopus_migration %>%
  mutate(
    iso2codefrom = countrycode(
      iso3codefrom,
      origin = "iso3c",
      destination = "iso2c"
    ),
    iso2codeto = countrycode(
      iso3codeto,
      origin = "iso3c",
      destination = "iso2c"
    )
  )

countries_locations <- countries %>%
  st_transform(8857) %>%
  st_point_on_surface() %>%
  st_transform(4326) %>%
  mutate(
    lon = st_coordinates(geom)[, 1],
    lat = st_coordinates(geom)[, 2]
  ) %>%
  st_drop_geometry() %>%
  select(
    name = iso_a2,
    x = lon,
    y = lat
  )
Warning: st_point_on_surface assumes attributes are constant over geometries

Mapping Scholarly Migration Flows

Aggregate migration flows between origin and destination countries across the study period.

Code 
scopus_flows <- scopus_migration %>%
  group_by(
    o = iso2codefrom,
    d = iso2codeto
  ) %>%
  summarise(
    value = sum(n_migrations, na.rm = TRUE),
    .groups = "drop"
  ) %>%
  filter(
    !is.na(o),
    !is.na(d),
    o != d
  ) %>%
  semi_join(countries_locations, by = c("o" = "name")) %>%
  semi_join(countries_locations, by = c("d" = "name"))

Country-Level Net Migration Balance

Calculate total incoming and outgoing scholarly migration flows and their net balance for each country.

Code 
country_flows <- bind_rows(
  scopus_flows %>%
    group_by(country = o) %>%
    summarise(outflows = sum(value), .groups = "drop"),

  scopus_flows %>%
    group_by(country = d) %>%
    summarise(inflows = sum(value), .groups = "drop")
) %>%
  group_by(country) %>%
  summarise(
    outflows = sum(outflows, na.rm = TRUE),
    inflows = sum(inflows, na.rm = TRUE),
    net_flow = inflows - outflows,
    .groups = "drop"
  )

country_flows <- country_flows %>%
  left_join(
    countries %>%
      st_drop_geometry() %>%
      select(country = iso_a2, country_name = name_long),
    by = "country"
  ) %>%
  select(country_name, inflows, outflows, net_flow)

Net Scholarly Migration by Country

The chart presents the 20 countries with the largest absolute net scholarly migration balances. Positive values indicate net receivers, while negative values indicate net senders.

Code 
top_net <- country_flows %>%
  arrange(desc(abs(net_flow))) %>%
  slice_head(n = 20) %>%
  mutate(
    country_name = reorder(country_name, net_flow)
  )

ggplot(
  top_net,
  aes(
    x = country_name,
    y = net_flow,
    fill = net_flow > 0
  )
) +
  geom_col() +
  coord_flip() +
  geom_hline(yintercept = 0, color = "black") +
  scale_fill_manual(
    values = c("TRUE" = "steelblue", "FALSE" = "indianred"),
    labels = c("FALSE" = "Net sender", "TRUE" = "Net receiver")
  ) +
  labs(
    title = "Net Scholarly Migration by Country",
    subtitle = "Aggregated Scopus flows across the study period",
    x = NULL,
    y = "Net migration balance (inflows - outflows)",
    fill = NULL
